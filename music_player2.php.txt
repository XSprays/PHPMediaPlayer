function getAlbumArt($albumDir, $musicDir) {
    if (!is_dir($musicDir . $albumDir)) return 'data:image/gif;base64,R0lGODlhAQABAIAAAP///wAAACH5BAEAAAAALAAAAAABAAEAAAICRAEAOw==';
    $extensions = ['jpg', 'jpeg', 'png', 'webp'];
    foreach ($extensions as $ext) {
        $cover = $musicDir . $albumDir . '/cover.' . $ext;
        if (file_exists($cover)) return 'music/' . ltrim($albumDir, '/\\') . '/cover.' . $ext . '?v=' . filemtime($cover);
    }
    $files = glob($musicDir . $albumDir . '/*.{jpg,jpeg,png,webp}', GLOB_BRACE);
    if ($files) {
        $f = $files[0];
        return 'music/' . ltrim($albumDir, '/\\') . '/' . basename($f) . '?v=' . filemtime($f);
    }
    return 'data:image/gif;base64,R0lGODlhAQABAIAAAP///wAAACH5BAEAAAAALAAAAAABAAEAAAICRAEAOw==';
}

// ── Load & parse tracks (unchanged logic) ───────────────────────────────────────────
$musicDir = __DIR__ . '/music/';
$musicFiles = [];

if (is_dir($musicDir) && is_readable($musicDir)) {
    $iterator = new RecursiveIteratorIterator(
        new RecursiveDirectoryIterator($musicDir, RecursiveDirectoryIterator::SKIP_DOTS),
        RecursiveIteratorIterator::LEAVES_ONLY
    );
    foreach ($iterator as $file) {
        if (strtolower($file->getExtension()) !== 'mp3') continue;
        $relPath = str_replace($musicDir, '', $file->getPathname());
        $relPath = ltrim($relPath, '/\\');

        $info = getMP3Info($file->getPathname()) ?: [
            'title'  => pathinfo($file->getFilename(), PATHINFO_FILENAME),
            'artist' => 'Unknown',
            'album'  => 'Unknown',
            'file'   => $relPath
        ];
        $info['file']     = $relPath;
        $info['album_art']= getAlbumArt(dirname($relPath) ?: '', $musicDir);
        $musicFiles[] = $info;
    }
    usort($musicFiles, fn($a,$b) => strcmp($a['title'], $b['title']));
}

// Unique artists & albums
$artists = array_unique(array_column($musicFiles, 'artist')); sort($artists);
$albums  = array_unique(array_column($musicFiles, 'album'));  sort($albums);

// ── Filtering ───────────────────────────────────────────────────────────────────────
$filterType  = null;
$filterValue = null;

    $filterType = 'artist'; $filterValue = $_GET['artist'];
} elseif (isset($_GET['album']) && in_array($_GET['album'], $albums)) {
    $filterType = 'album'; $filterValue = $_GET['album'];
}

$filteredTracks = $musicFiles;
if ($filterType) {
    $filteredTracks = array_filter($musicFiles, fn($m) => $m[$filterType] === $filterValue);
}

// ── Pagination ──────────────────────────────────────────────────────────────────────
$tracksPerPage = 20;
$page = isset($_GET['page']) ? max(1, (int)$_GET['page']) : 1;
$offset = ($page - 1) * $tracksPerPage;
$totalTracks = count($filteredTracks);
$totalPages = max(1, ceil($totalTracks / $tracksPerPage));
$pagedTracks = array_slice($filteredTracks, $offset, $tracksPerPage);

// ── Selected track ──────────────────────────────────────────────────────────────────

    $track = $allFiles[0] ?? '';
}
